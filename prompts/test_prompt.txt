Given a function signature and behavior, produce 
complete unit tests that check:

1) HAPPY PATH TESTS:
   - Normal expected inputs
   - Typical use cases
   - Expected return values

2) EDGE CASE TESTS:
   - Empty inputs (empty string, empty list, zero)
   - Null/None/nil inputs
   - Boundary values (min, max)
   - Very large inputs
   - Special characters in strings

3) ERROR/EXCEPTION TESTS:
   - Invalid input types
   - Out of range values
   - Division by zero scenarios
   - Null pointer scenarios
   - Index out of bounds scenarios

4) SECURITY TESTS:
   - SQL injection attempts
   - Command injection attempts
   - Extremely long inputs
   - Special character inputs

TEST STRUCTURE BY LANGUAGE:

Python (pytest):
\`\`\`python
import pytest

def test_function_normal_case():
    assert function(input) == expected

def test_function_empty_input():
    assert function([]) == []

def test_function_raises_on_none():
    with pytest.raises(TypeError):
        function(None)

def test_function_zero_division():
    with pytest.raises(ZeroDivisionError):
        function(0)
\`\`\`

Java (JUnit 5):
\`\`\`java
import org.junit.jupiter.api.*;
import static org.junit.jupiter.api.Assertions.*;

class FunctionTest {
    @Test
    void testNormalCase() {
        assertEquals(expected, function(input));
    }

    @Test
    void testNullInput() {
        assertThrows(NullPointerException.class, 
            () -> function(null));
    }

    @Test
    void testEmptyInput() {
        assertEquals(0, function(new int[]{}));
    }
}
\`\`\`

JavaScript (Jest):
\`\`\`javascript
describe('functionName', () => {
    test('normal case', () => {
        expect(function(input)).toBe(expected);
    });

    test('null input', () => {
        expect(() => function(null)).toThrow();
    });

    test('empty array', () => {
        expect(function([])).toEqual([]);
    });
});
\`\`\`

Go (testing package):
\`\`\`go
func TestFunctionNormalCase(t *testing.T) {
    result := Function(input)
    if result != expected {
        t.Errorf("Expected %v got %v", expected, result)
    }
}

func TestFunctionEdgeCase(t *testing.T) {
    result := Function(nil)
    if result != nil {
        t.Errorf("Expected nil got %v", result)
    }
}
\`\`\`
